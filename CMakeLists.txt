cmake_minimum_required(VERSION 3.20)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" AULA_VERSION LIMIT_COUNT 1)
project(AulaControl VERSION ${AULA_VERSION} LANGUAGES C CXX)

if(NOT WIN32)
  message(FATAL_ERROR "Este proyecto de diagnóstico está destinado a Windows.")
endif()
enable_language(RC)

include(FetchContent)
set(HIDAPI_WITH_TESTS OFF CACHE BOOL "" FORCE)
set(HIDAPI_WITH_HIDTEST OFF CACHE BOOL "" FORCE)
FetchContent_Declare(hidapi
  GIT_REPOSITORY https://github.com/libusb/hidapi.git
  GIT_TAG hidapi-0.15.0
  GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(hidapi)

add_executable(hid_scanner src/main.cpp)
target_compile_features(hid_scanner PRIVATE cxx_std_17)
target_link_libraries(hid_scanner PRIVATE hidapi::hidapi)
if(MSVC)
  target_compile_options(hid_scanner PRIVATE /W4 /utf-8)
else()
  target_compile_options(hid_scanner PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_custom_command(TARGET hid_scanner POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_if_different
    $<TARGET_FILE:hidapi::hidapi> $<TARGET_FILE_DIR:hid_scanner>)
add_executable(bateria_f75 src/battery.cpp)
target_compile_features(bateria_f75 PRIVATE cxx_std_17)
target_link_libraries(bateria_f75 PRIVATE hidapi::hidapi)
if(MSVC)
  target_compile_options(bateria_f75 PRIVATE /W4 /utf-8)
endif()
add_custom_command(TARGET bateria_f75 POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_if_different
    $<TARGET_FILE:hidapi::hidapi> $<TARGET_FILE_DIR:bateria_f75>)
add_executable(rgb_preview src/rgb_preview.cpp)
target_compile_features(rgb_preview PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(rgb_preview PRIVATE /W4 /utf-8)
endif()
add_executable(rgb_prueba src/rgb_test.cpp)
target_compile_features(rgb_prueba PRIVATE cxx_std_17)
target_link_libraries(rgb_prueba PRIVATE hidapi::hidapi)
if(MSVC)
  target_compile_options(rgb_prueba PRIVATE /W4 /utf-8)
endif()
add_custom_command(TARGET rgb_prueba POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_if_different
    $<TARGET_FILE:hidapi::hidapi> $<TARGET_FILE_DIR:rgb_prueba>)
set(RAVEN_BUILD_WEB OFF CACHE BOOL "" FORCE)
add_subdirectory(third_party/Raven-0.2.3 EXCLUDE_FROM_ALL)
configure_file(resources/aula_control.rc.in "${CMAKE_CURRENT_BINARY_DIR}/aula_control.rc" @ONLY)
set_source_files_properties("${CMAKE_CURRENT_BINARY_DIR}/aula_control.rc" PROPERTIES
  OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/resources/aula_control.ico")
add_executable(aula_control WIN32 src/gui.cpp "${CMAKE_CURRENT_BINARY_DIR}/aula_control.rc")
target_compile_features(aula_control PRIVATE cxx_std_17)
target_include_directories(aula_control PRIVATE third_party/Raven-0.2.3/bindings/cpp)
target_link_libraries(aula_control PRIVATE raven::raven user32 shell32 advapi32 gdi32)
if(MSVC)
  target_compile_options(aula_control PRIVATE /W4 /utf-8)
  target_link_options(aula_control PRIVATE /ENTRY:mainCRTStartup)
endif()
add_dependencies(aula_control rgb_prueba bateria_f75)
